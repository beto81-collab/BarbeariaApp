rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    
    // ==================== REGRAS DE USUÁRIOS ====================
    match /usuarios/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
      allow create: if request.auth != null && request.auth.uid == userId;
    }
    

    
    // ==================== REGRAS DE SERVIÇOS ====================
    match /servicos/{servicoId} {
      allow read, write: if true;
      allow write: if request.auth != null && 
        isAdmin(request.auth.uid);
    }
    
    // ==================== REGRAS DE PRODUTOS ====================
    match /produtos/{produtoId} {
      allow read, write: if true;
      allow write: if request.auth != null && 
        isAdmin(request.auth.uid);
    }

    // ==================== REGRAS DE CONFIG (HORÁRIOS, ETC) ====================
    match /config/{document} {
      allow read, write: if request.auth != null && isAdmin(request.auth.uid);
    }
    
    // ==================== FUNÇÕES AUXILIARES ====================
    function isAdmin(userId) {
      return exists(/databases/$(database)/documents/usuarios/$(userId)) &&
        get(/databases/$(database)/documents/usuarios/$(userId)).data.tipo == 'admin';
    }
    function isBarbeiro(userId) {
      return exists(/databases/$(database)/documents/usuarios/$(userId)) &&
        get(/databases/$(database)/documents/usuarios/$(userId)).data.tipo == 'barbeiro';
    }
    function validateAgendamento() {
      let agendamento = request.resource.data;
      return 
        agendamento.clienteId == request.auth.uid &&
        agendamento.keys().hasAll(['clienteId', 'barbeiroId', 'servicoId', 'dataHora', 'status', 'valor']) &&
        agendamento.clienteId is string &&
        agendamento.barbeiroId is string &&
        agendamento.servicoId is string &&
        agendamento.dataHora is timestamp &&
        agendamento.status is string &&
        agendamento.valor is number &&
        agendamento.valor > 0 &&
        agendamento.dataHora > request.time &&
        agendamento.status == 'agendado';
    }
  }
}
